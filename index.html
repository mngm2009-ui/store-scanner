async function openScanner() { if (!audioCtx) audioCtx = new (window.AudioContext || window.webkitAudioContext)(); beepSuccess.play().then(() => { beepSuccess.pause(); beepSuccess.currentTime = 0; }).catch(() => {}); document.getElementById('camera-wrapper').style.display = 'block'; document.getElementById('btn-start').style.display = 'none'; // إذا كان الآيفون يدعم القارئ الذكي المدمج (iOS الحديث) if ('BarcodeDetector' in window) { try { const detector = new BarcodeDetector({ formats: ['ean_13', 'ean_8', 'upc_a', 'code_128', 'code_39', 'qr_code'] }); const stream = await navigator.mediaDevices.getUserMedia({ video: { facingMode: "environment", width: { ideal: 1280 } } }); const readerContainer = document.getElementById('reader'); readerContainer.innerHTML = ''; const videoEl = document.getElementById('ios-video'); videoEl.srcObject = stream; await videoEl.play(); isCameraRunning = true; const scanTimer = setInterval(async () => { if (!isCameraRunning) { clearInterval(scanTimer); stream.getTracks().forEach(t => t.stop()); return; } try { const barcodes = await detector.detect(videoEl); if (barcodes.length > 0) { clearInterval(scanTimer); stream.getTracks().forEach(t => t.stop()); closeScanner(); handleScanResult(barcodes[0].rawValue); } } catch (e) {} }, 150); return; } catch (err) { console.warn("جاري التبديل للمحرك البديل...", err); } } // في حال عدم دعمه، يتم استخدام المكتبة العادية if (!html5QrCode) { html5QrCode = new Html5Qrcode("reader"); } const config = { fps: 20, qrbox: { width: 280, height: 160 } }; html5QrCode.start({ facingMode: "environment" }, config, onScanSuccess) .then(() => { isCameraRunning = true; }) .catch(err => { alert("تعذر تشغيل الكاميرا: " + err); closeScanner(); }); }
